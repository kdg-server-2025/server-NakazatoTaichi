## 6回目の授業
```bash
# レジストリミラーにmirror.gcr.ioを設定をする
# 参考記事
https://zenn.dev/hosht/articles/29d9f9f698dd5a

# dockerのシステム情報を確認
docker system info

# brewでaquaをインストールする
brew install aqua

# サーバー構築のリポジトリのディレクトリに移動して
aqua init

# 以下のコマンドを実行して、 kubernetes-sigs/kindのツールを選択する
aqua g -i

# 追加したツールを使えるようにする
aqua i -l

# kindのパスを通す
export PATH="${AQUA_ROOT_DIR:-${XDG_DATA_HOME:-$HOME/.local/share}/aquaproj-aqua}/bin:$PATH"

# インストールされていることを確認
kind version

# cluster.yamlを作成
# HA 構成の Cluster を作成する

# config を指定して cluster を作成
kind create cluster --config cluster.yaml

# DOCKER_CONTENT_TRUSTを一時的に無効にした
DOCKER_CONTENT_TRUST=0

# 参考記事
https://qiita.com/railgun-0402/items/5a0fd3a4336c2c8ba428

# aquaでkubectlとk9sを使えるようにする
# kubernetes/kubernetes/kubectlとderailed/k9sを選択して使えるようにする
aqua g -i
aqua i -l

# k8sのnode一覧を表示
kubectl get nodes

# aquaでhelm/helmを選択して使えるようにする
aqua g -i
aqua i -l

# Helmのリポジトリを追加
helm repo add bitnami https://charts.bitnami.com/bitnami
helm repo update

# Helm chartをk8sクラスタに適用
helm install kdg-server-nginx bitnami/nginx

# ブラウザから nginx にアクセスできるようにポートフォワードする
kubectl port-forward service/kdg-server-nginx 8080:80
```
